    ```python
    @router.get("/licenses/{license_id}", response_class=HTMLResponse)
    def license_detail(license_id: int, request: Request, db: SessionDep):
        license = db.get(License, license_id)
        if not license:
            return templates.TemplateResponse(
                request=request,
                name="404.html",
                context={"title": "Лицензия не найдена"},
                status_code=404,
            )
        threshold = get_global_threshold(db)
        effective_threshold = license.notify_days_before if license.notify_days_before is not None else threshold
        days_left = days_until_expiry(license.expiry_date)
        status = get_license_status(license.expiry_date, effective_threshold)
        return templates.TemplateResponse(
            request=request,
            name="license_detail.html",
            context={
                "license": license,
                "status": status,
                "days_left": days_left,
                "title": license.product_name,
            },
        )
    ```

    **IMPORTANT:** This endpoint MUST be placed BEFORE `GET /licenses/new` in the router file, OR use a different route pattern. Since FastAPI matches routes in order, `GET /licenses/{license_id}` will intercept `GET /licenses/new` if placed first. To avoid this, place `GET /licenses/new` BEFORE `GET /licenses/{license_id}` in the file. The executor should verify route ordering.
